cmake_minimum_required(VERSION 3.16)
project(SPARK VERSION 0.13.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MariaDB REQUIRED IMPORTED_TARGET libmariadb)
add_library(spark_core src/api.cpp src/devices.cpp src/database.cpp src/timers.cpp src/notifications.cpp src/pulse.cpp src/calendar.cpp src/daily_tasks.cpp src/files.cpp)
target_compile_features(spark_core PUBLIC cxx_std_17)
target_include_directories(spark_core PUBLIC src)
target_include_directories(spark_core SYSTEM PUBLIC third_party third_party/cpp-httplib)
target_compile_definitions(spark_core PUBLIC SPARK_VERSION="${PROJECT_VERSION}")
target_link_libraries(spark_core PUBLIC Threads::Threads PkgConfig::MariaDB)
add_executable(spark_server src/main.cpp)
target_link_libraries(spark_server PRIVATE spark_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(file_tests tests/file_test.cpp)
  target_link_libraries(file_tests PRIVATE spark_core)
  add_test(NAME file_storage COMMAND file_tests)
  set_tests_properties(file_storage PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77 DEPENDS daily_tasks)
  add_executable(spark_tests tests/api_test.cpp)
  target_link_libraries(spark_tests PRIVATE spark_core)
  add_test(NAME api_contract COMMAND spark_tests)
  add_executable(notification_tests tests/notification_test.cpp)
  target_link_libraries(notification_tests PRIVATE spark_core)
  add_test(NAME tcp_notifications COMMAND notification_tests)
  set_tests_properties(tcp_notifications PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 DEPENDS timer_lifecycle)
  add_executable(calendar_tests tests/calendar_test.cpp)
  target_link_libraries(calendar_tests PRIVATE spark_core)
  add_test(NAME calendar_events COMMAND calendar_tests)
  set_tests_properties(calendar_events PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 DEPENDS tcp_notifications)
  add_executable(daily_task_tests tests/daily_task_test.cpp)
  target_link_libraries(daily_task_tests PRIVATE spark_core)
  add_test(NAME daily_tasks COMMAND daily_task_tests)
  set_tests_properties(daily_tasks PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 DEPENDS calendar_events)
  add_executable(timer_tests tests/timer_test.cpp)
  target_link_libraries(timer_tests PRIVATE spark_core)
  add_test(NAME timer_lifecycle COMMAND timer_tests)
  set_tests_properties(timer_lifecycle PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 DEPENDS api_contract)
  set_tests_properties(api_contract PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
