cmake_minimum_required(VERSION 3.16)
project(SPARK VERSION 0.18.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MariaDB REQUIRED IMPORTED_TARGET libmariadb)
add_library(spark_core src/credential_management.cpp src/auth.cpp src/browser_sessions.cpp src/browser_auth.cpp src/authorization.cpp src/config.cpp src/api.cpp src/commands.cpp src/dashboard.cpp src/devices.cpp src/database.cpp src/timers.cpp src/notifications.cpp src/pulse.cpp src/calendar.cpp src/daily_tasks.cpp src/files.cpp src/projects.cpp src/boards.cpp)
target_compile_features(spark_core PUBLIC cxx_std_17)
target_include_directories(spark_core PUBLIC src)
target_include_directories(spark_core SYSTEM PUBLIC third_party third_party/cpp-httplib)
target_compile_definitions(spark_core PRIVATE SPARK_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(spark_core PUBLIC SPARK_VERSION="${PROJECT_VERSION}")
target_link_libraries(spark_core PUBLIC Threads::Threads PkgConfig::MariaDB OpenSSL::Crypto)
add_executable(spark_server src/main.cpp)
target_link_libraries(spark_server PRIVATE spark_core)
add_executable(spark_credentials src/auth_main.cpp)
target_link_libraries(spark_credentials PRIVATE spark_core)
include(CTest)
